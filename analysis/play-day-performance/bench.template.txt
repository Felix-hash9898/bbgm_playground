function wrap(obj: any, key: string, name: string) {
 if (typeof obj[key] !== "function") return;
 const original = vi.isMockFunction(obj[key]) ? obj[key].getMockImplementation()! : obj[key];
 vi.spyOn(obj, key).mockImplementation(function(this: any, ...args: any[]) {
  if (!active) return original.apply(this,args);

   const result = original.apply(this,args);
   if (result && typeof result.then === "function") return result.finally(() => record(name, performance.now()-start));

 });
}
const seedRandom = (seed: number) => () => {

 t = Math.imul(t ^ t >>> 15, t | 1);
 t ^= t + Math.imul(t ^ t >>> 7, t | 61);
 return ((t ^ t >>> 14) >>> 0) / 4294967296;
};
